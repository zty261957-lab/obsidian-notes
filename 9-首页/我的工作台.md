## 今日新增 / 修改

```base
filters:
  and:
    - 'file.ext == "md"'
    - 'file.path != this.file.path'
    - '!file.inFolder("9-首页/模板")'
    - or:
        - 'file.ctime.date().toString() == today().toString()'
        - 'file.mtime.date().toString() == today().toString()'
formulas:
  change_type: 'if(file.ctime.date().toString() == today().toString(), "新增", "修改")'
  changed_at: 'file.mtime.format("HH:mm")'
properties:
  file.name:
    displayName: 笔记
  formula.change_type:
    displayName: 类型
  formula.changed_at:
    displayName: 时间
views:
  - type: cards
    name: 今日动态
    order:
      - file.name
      - formula.change_type
      - formula.changed_at
    sort:
      - property: file.mtime
        direction: DESC
    limit: 10
```
